% count positive numbers
x1 = 0          % address of N
x2 = [x1]       % N
x3 = 4          % array address
x4 = 0          % index
x5 = 0          % positive count

Print x2        % Print array size N
Print x5        % Print initial positive count (0)

.loop
x6 = x2 - x4
BEQ .exit
x7 = [x3]
x8 = x7 - 0
BLT .skip       % Skip if number is negative (< 0)
x5 = x5 + 1     % Increment positive count
.skip
x3 = x3 + 4
x4 = x4 + 1
BAL .loop
.exit
x1 = 16
[x1] = x5
Print x5        % Print final positive count
Print x1        % Print target memory destination (16)
