% Program to find maximum number
x1 = 0          % address of N
x2 = [x1]       % N

x3 = 4          % current array address
x4 = 0          % index
x5 = [x3]       % current maximum = first element

x3 = x3 + 4
x4 = x4 + 1

.loop
x6 = x2 - x4
BEQ .exit

x7 = [x3]       % current element

x8 = x7 - x5    % current - maximum
BMI .skip       % if current < maximum, don't update

x5 = x7         % maximum = current

.skip
x3 = x3 + 4
x4 = x4 + 1
BAL .loop

.exit
[x1] = x5
