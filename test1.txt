% Program to find maximum number
x1 = 0           % address of N
x2 = [x1]        % N (array size)

x3 = 4           % current array address
x4 = 0           % index counter
x5 = [x3]        % current maximum = first element

% Log initial values (Array size and initial max)
Print x2         % Print total elements N
Print x5         % Print initial maximum element

x3 = x3 + 4
x4 = x4 + 1
.loop
x6 = x2 - x4
BEQ .exit
x7 = [x3]        % current element
x8 = x7 - x5     % current - maximum
BMI .skip        % if current < maximum, skip update
x5 = x7          % update maximum = current
.skip
x3 = x3 + 4
x4 = x4 + 1
BAL .loop
.exit
[x1] = x5         % Write global maximum back to memory address 0
Print x5         % Log final maximum result
Print x1         % Log target memory location
