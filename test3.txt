% 3-tap Moving Average / Convolution Filter ([1, 2, 1])
x1 = 0
x2 = [x1]        % N (total input elements)

x3 = 4          % input array starting address
x4 = 32         % output array starting address
x5 = 0          % iteration counter

x6 = x2 - 2     % total output size (N - 2)

Print x2        % Print array size N
Print x6        % Print number of filter windows to compute

.loop
x7 = x6 - x5
BEQ .exit

x8 = [x3]       % window[0]

x9 = x3 + 4
x9 = [x9]       % window[1]

x10 = x3 + 8
x10 = [x10]     % window[2]

x11 = x8 * 1     % apply weight 1
x12 = x9 * 2     % apply weight 2
x13 = x10 * 1    % apply weight 1

x14 = x11 + x12
x14 = x14 + x13  % x14 = filtered value

[x4] = x14      % store result to output array

Print x5        % Print current window index
Print x14       % Print computed filtered result
Print x4        % Print destination memory address

x3 = x3 + 4     % slide window right by 1 element
x4 = x4 + 4     % move to next output slot
x5 = x5 + 1
BAL .loop

.exit
Print x5        % Print total processed filter steps
